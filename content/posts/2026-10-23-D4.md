; ---- 三魔神 ----
F4::
{
    BARS := [
        { name: "进度条1", x: 100, y: 200 },
        { name: "进度条2", x: 300, y: 200 },
        { name: "进度条3", x: 500, y: 200 }
    ]

    EMPTY_COLOR := 0x2B2B2B   ; 空（未填充）时该像素的颜色
    TOLERANCE   := 12         ; 每通道允许色差，0 = 完全一致

    ColorNear(a, b, tol) {
        return Abs(((a >> 16) & 0xFF) - ((b >> 16) & 0xFF)) <= tol
            && Abs(((a >> 8)  & 0xFF) - ((b >> 8)  & 0xFF)) <= tol
            && Abs(( a        & 0xFF) - ( b        & 0xFF)) <= tol
    }

    IsEmpty(bar) {
        return ColorNear(Integer(PixelGetColor(bar.x, bar.y)), EMPTY_COLOR, TOLERANCE)
    }

    AllEmpty() {
        for bar in BARS
            if !IsEmpty(bar)
                return false
        return true
    }
    Loop {
        if AllEmpty() {
                    SendEvent "t"
                    Sleep 500
                    SendEvent "t"
                    Sleep 500
                    SendEvent "t"
                    Break
                }
