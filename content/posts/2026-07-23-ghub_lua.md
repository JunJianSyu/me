-- 办公重复操作脚本（Logitech G HUB）
-- 触发方式：按下鼠标侧键（按钮4）执行整套流程
-- 坐标系统：基于 100x100 网格，与分辨率无关

-- ====== 唯一需要修改的地方：你的屏幕分辨率 ======
local SCREEN_W = 1920
local SCREEN_H = 1080

-- 将百分比坐标(0~100)转换为实际像素坐标
local function pos(x, y)
    return math.floor(SCREEN_W * x / 100), math.floor(SCREEN_H * y / 100)
end

-- 点击指定百分比坐标
local function click_at(x, y)
    local px, py = pos(x, y)
    MoveMouseTo(px, py)
    Sleep(100)
    PressMouseButton(1)
    Sleep(50)
    ReleaseMouseButton(1)
    Sleep(200)
end

-- 按下并释放一个键
local function tap_key(key)
    PressKey(key)
    Sleep(50)
    ReleaseKey(key)
    Sleep(80)
end

function OnEvent(event, arg)
    if event == "MOUSE_BUTTON_PRESSED" and arg == 4 then

        -- 步骤1：点击屏幕偏左上位置（宽度的26%，高度的14%）
        -- 原 1920x1080 下对应 (500, 150)，换分辨率无需改动
        click_at(26, 14)

        -- 步骤2：点击屏幕正中间
        click_at(50, 50)

        -- 步骤3：按键 5、4、3、2 循环4次
        for i = 1, 4 do
            tap_key("5")
            tap_key("4")
            tap_key("3")
            tap_key("2")
        end

    end
end
